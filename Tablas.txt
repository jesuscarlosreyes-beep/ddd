    id_alumno STRING,
    curp_alumno STRING,
    nombres_alumno STRING,
    apellido1_alumno STRING,
    apellido2_alumno STRING,
    nombre_completo_alumno STRING,
    matricula_alumno STRING

    "separatorChar" = ",",
    "quoteChar" = "\"",
    "escapeChar" = "\\"
)
